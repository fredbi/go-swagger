The command generates (or regenerates) selected skeleton server-side handlers with the appropriate parameters and responses.

The generated code takes care of the parameters and response binding and the `ServeHTTP` boiler plate: your handler
is a responder (optionally strongly typed) with input parameters as a go type.

### Options

[Common options](./reference/codegen_options)

* `--skip-tag-packages`: do not construct packages per operation tag (which is the default)
* `--tags`: filter to generate only operations with the specified tags
* `--api-package`: target folder beneath `./restapi` (`--server-package`) to hold operations
* `--with-enum-ci`: generates case insensitive enum validations only for operation parameters (and response headers)

### Example

Several complete commented examples of generated servers are presented [on the examples doc site][server-example].

[server-example]: https://goswagger.io/examples/guides/servers/

```cmd
swagger generate operation --api-package handlers --skip-tag-packages --operation getPets ./testdata/petstores/petstore.json
```

```go
// Code generated by go-swagger; DO NOT EDIT.

package handlers

import (
	"net/http"

	"github.com/go-openapi/runtime/middleware"
)

// GetPetsHandlerFunc turns a function with the right signature into a get pets handler
type GetPetsHandlerFunc func(GetPetsParams) middleware.Responder

// Handle executing the request and returning a response
func (fn GetPetsHandlerFunc) Handle(params GetPetsParams) middleware.Responder {
	return fn(params)
}

// GetPetsHandler interface for that can handle valid get pets params
type GetPetsHandler interface {
	Handle(GetPetsParams) middleware.Responder
}

// NewGetPets creates a new http.Handler for the get pets operation
func NewGetPets(ctx *middleware.Context, handler GetPetsHandler) *GetPets {
	return &GetPets{Context: ctx, Handler: handler}
}

// GetPets swagger:route GET /pets Pet Operations getPets
//
// finds pets in the system
type GetPets struct {
	Context *middleware.Context
	Handler GetPetsHandler
}

func (o *GetPets) ServeHTTP(rw http.ResponseWriter, r *http.Request) {
	route, rCtx, _ := o.Context.RouteInfo(r)
	if rCtx != nil {
		*r = *rCtx
	}
	params := NewGetPetsParams()
	if err := o.Context.BindValidRequest(r, route, &params); err != nil { // bind params
		o.Context.Respond(rw, r, route.Produces, route, err)
		return
	}

	res := o.Handler.Handle(params) // actually handle the request

	o.Context.Respond(rw, r, route.Produces, route, res)

}
```
